\begin{answer}
\textbf{Loss function.} The NLL for a single example is:
\[
\ell(\theta) = -\log p(y;\ \eta) = a(\eta) - \eta y - \log b(y), \quad \eta = \theta^T x.
\]
Over $n$ examples: $L(\theta) = \frac{1}{n}\sum_i \bigl[a(\theta^T x^{(i)}) - (\theta^T x^{(i)})y^{(i)} - \log b(y^{(i)})\bigr]$.

\textbf{Gradient.}
\[
\frac{\partial L}{\partial \theta_j} = \frac{1}{n}\sum_i \bigl(a'(\eta^{(i)}) - y^{(i)}\bigr)\,x_j^{(i)}
= \frac{1}{n}\sum_i \bigl(\mathbb{E}[Y^{(i)};\eta^{(i)}] - y^{(i)}\bigr)\,x_j^{(i)}.
\]

\textbf{Hessian.} Differentiating once more:
\[
H_{jk} = \frac{\partial^2 L}{\partial\theta_j\partial\theta_k}
= \frac{1}{n}\sum_i a''(\eta^{(i)})\,x_j^{(i)} x_k^{(i)}.
\]
In matrix form: $H = \frac{1}{n}\sum_i a''(\eta^{(i)})\,x^{(i)}{x^{(i)}}^T$.

\textbf{$H$ is PSD.} For any vector $z$:
\[
z^T H z = \frac{1}{n}\sum_i a''(\eta^{(i)})\,(x^{(i)T}z)^2.
\]
By part (b), $a''(\eta) = \operatorname{Var}(Y;\eta) \ge 0$, and $(x^{(i)T}z)^2 \ge 0$. Hence every term is non-negative, so $z^T H z \ge 0$ for all $z$, i.e., $H \succeq 0$ and the NLL of any GLM is convex. $\blacksquare$
\end{answer}

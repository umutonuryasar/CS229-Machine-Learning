\begin{answer}

\begin{enumerate}

\item \textbf{$K(x,z) = K_1(x,z) + K_2(x,z)$: Yes, it is a kernel.}

Let $K$, $K_1$, $K_2$ be the $n\times n$ kernel matrices for any finite set $\{x^{(1)},\ldots,x^{(n)}\}$. Then $K = K_1 + K_2$.
\begin{itemize}
  \item \emph{Symmetry:} $K_{ij} = K_1(x^{(i)},x^{(j)}) + K_2(x^{(i)},x^{(j)}) = K_{ji}$.
  \item \emph{PSD:} For any $z\in\mathbb{R}^n$, $z^T K z = z^T K_1 z + z^T K_2 z \geq 0$ since $K_1$ and $K_2$ are PSD.
\end{itemize}

\item \textbf{$K(x,z) = K_1(x,z) - K_2(x,z)$: Not necessarily a kernel.}

\emph{Counterexample:} Let $K_1(x,z) = 0$ (the zero kernel, which is valid and PSD) and $K_2(x,z) = x^Tz$ (the linear kernel, which is valid). Then $K(x,z) = -x^Tz$. For the single-point set $\{x^{(1)}=[1]\}$, the kernel matrix is $[-1]$, which is negative definite and hence not PSD.

\item \textbf{$K(x,z) = aK_1(x,z)$ with $a > 0$: Yes, it is a kernel.}

For any $z\in\mathbb{R}^n$, $z^T(aK_1)z = a\cdot z^TK_1 z \geq 0$ since $a > 0$ and $K_1$ is PSD. Symmetry is immediate.

\item \textbf{$K(x,z) = -aK_1(x,z)$ with $a > 0$: Not necessarily a kernel.}

\emph{Counterexample:} Let $K_1(x,z) = x^Tz$. For $\{x^{(1)}=[1]\}$, the kernel matrix of $K_1$ is $[1]$, giving kernel matrix of $K$ equal to $[-a] < 0$. Not PSD.

\item \textbf{$K(x,z) = K_1(x,z)K_2(x,z)$: Yes, it is a kernel.}

We prove the Schur (Hadamard) product of two PSD matrices is PSD. Let $M_1$ and $M_2$ be the kernel matrices (both symmetric PSD). Write $M_1 = \sum_k \lambda_k u_k u_k^T$ (eigendecomposition, $\lambda_k \geq 0$). Then:
$$z^T(M_1 \circ M_2)z = \sum_{i,j} z_i z_j (M_1)_{ij}(M_2)_{ij} = \sum_k \lambda_k \sum_{i,j} z_i (u_k)_i \,(M_2)_{ij}\, z_j (u_k)_j = \sum_k \lambda_k\, w_k^T M_2\, w_k,$$
where $w_k \in \mathbb{R}^n$ has $i$-th component $(w_k)_i = z_i (u_k)_i$. Since $M_2$ is PSD, $w_k^T M_2 w_k \geq 0$ for all $k$. And $\lambda_k \geq 0$, so the entire sum is $\geq 0$. Therefore $M_1 \circ M_2$ is PSD, meaning $K = K_1 K_2$ is a valid kernel.

\item \textbf{$K(x,z) = f(x)f(z)$: Yes, it is a kernel.}

The kernel matrix has entries $K_{ij} = f(x^{(i)})f(x^{(j)})$. Let $\mathbf{f} = [f(x^{(1)}),\ldots,f(x^{(n)})]^T$; then $K = \mathbf{f}\mathbf{f}^T$. For any $z$: $z^T K z = z^T \mathbf{f}\mathbf{f}^T z = (\mathbf{f}^T z)^2 \geq 0$. Symmetry is clear.

Alternatively, this is an inner product in $\mathbb{R}^1$ via $\phi(x) = f(x)$.

\item \textbf{$K(x,z) = K_3(\phi(x), \phi(z))$: Yes, it is a kernel.}

The kernel matrix for this $K$ evaluated on $\{x^{(1)},\ldots,x^{(n)}\}$ is identical to the kernel matrix for $K_3$ evaluated on $\{\phi(x^{(1)}),\ldots,\phi(x^{(n)})\}$. Since $K_3$ is a valid kernel over $\mathbb{R}^p\times\mathbb{R}^p$, that matrix is symmetric PSD for any finite set of points in $\mathbb{R}^p$. Hence $K$ is a valid kernel.

\item \textbf{$K(x,z) = p(K_1(x,z))$ where $p$ has positive coefficients: Yes, it is a kernel.}

Write $p(t) = \sum_{m=0}^{M} c_m t^m$ with $c_m \geq 0$. We use the closure properties proved above:
\begin{itemize}
  \item $K_1^m := \underbrace{K_1 \cdots K_1}_{m \text{ times}}$ is a kernel by repeated application of part (e).
  \item $c_m K_1^m$ is a kernel for each $m$ by part (c) (since $c_m \geq 0$; for $c_m = 0$ the zero kernel is trivially PSD).
  \item $p(K_1) = \sum_m c_m K_1^m$ is a kernel by repeated application of part (a).
\end{itemize}

\end{enumerate}

\end{answer}

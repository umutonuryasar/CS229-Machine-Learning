\begin{answer}
We show $\ell_\text{semi-sup}(\theta^{(t+1)}) \geq \ell_\text{semi-sup}(\theta^{(t)})$.

Recall the standard EM lower bound (ELBO). For any distribution $Q_i$ over $\zsi$:
$$\ell_\text{unsup}(\theta) = \sum_{i=1}^\nexp \log p(\xsi;\theta) \geq \sum_{i=1}^\nexp \sum_{\zsi} Q_i(\zsi)\log\frac{p(\xsi,\zsi;\theta)}{Q_i(\zsi)} =: \mathcal{L}(\theta, Q).$$

Equality holds when $Q_i(\zsi) = p(\zsi|\xsi;\theta)$.

\textbf{E-step:} We choose $Q_i^{(t)}(\zsi) = p(\zsi|\xsi;\theta^{(t)})$, so:
$$\mathcal{L}(\theta^{(t)}, Q^{(t)}) = \ell_\text{unsup}(\theta^{(t)}).$$

\textbf{M-step:} We have
$$\theta^{(t+1)} = \arg\max_\theta \left[\mathcal{L}(\theta, Q^{(t)}) + \alpha\,\ell_\text{sup}(\theta)\right],$$
so by the maximality of $\theta^{(t+1)}$:
$$\mathcal{L}(\theta^{(t+1)}, Q^{(t)}) + \alpha\,\ell_\text{sup}(\theta^{(t+1)}) \geq \mathcal{L}(\theta^{(t)}, Q^{(t)}) + \alpha\,\ell_\text{sup}(\theta^{(t)}).$$

Combining with the ELBO inequality $\ell_\text{unsup}(\theta^{(t+1)}) \geq \mathcal{L}(\theta^{(t+1)}, Q^{(t)})$:
\begin{align*}
\ell_\text{semi-sup}(\theta^{(t+1)})
&= \ell_\text{unsup}(\theta^{(t+1)}) + \alpha\,\ell_\text{sup}(\theta^{(t+1)}) \\
&\geq \mathcal{L}(\theta^{(t+1)}, Q^{(t)}) + \alpha\,\ell_\text{sup}(\theta^{(t+1)}) \\
&\geq \mathcal{L}(\theta^{(t)}, Q^{(t)}) + \alpha\,\ell_\text{sup}(\theta^{(t)}) \\
&= \ell_\text{unsup}(\theta^{(t)}) + \alpha\,\ell_\text{sup}(\theta^{(t)}) \\
&= \ell_\text{semi-sup}(\theta^{(t)}). \qquad \square
\end{align*}
\end{answer}

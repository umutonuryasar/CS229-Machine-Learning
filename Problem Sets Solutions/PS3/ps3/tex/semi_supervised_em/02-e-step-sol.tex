\begin{answer}
\textbf{Latent variables:} The latent variables are $\zsi \in \{1,\ldots,k\}$ for each unlabeled example $\xsi$, $i = 1,\ldots,\nexp$. The labels $\tilde{z}^{(i)}$ of the labeled examples are \emph{observed}, so they are not re-estimated in the E-step.

\textbf{E-step update:} For each unlabeled example $\xsi$ and cluster $j$, set
$$w_{ij} := p(\zsi = j \mid \xsi;\, \mu, \Sigma, \phi)
= \frac{p(\xsi \mid \zsi = j;\, \mu_j, \Sigma_j)\, p(\zsi = j;\, \phi_j)}{\displaystyle\sum_{l=1}^k p(\xsi \mid \zsi = l;\, \mu_l, \Sigma_l)\, p(\zsi = l;\, \phi_l)}.$$

Substituting the Gaussian likelihood $p(\xsi \mid \zsi = j) = \mathcal{N}(\xsi;\, \mu_j, \Sigma_j)$:

$$\boxed{w_{ij} = \frac{\phi_j \,\mathcal{N}(\xsi;\,\mu_j,\Sigma_j)}{\displaystyle\sum_{l=1}^k \phi_l\, \mathcal{N}(\xsi;\,\mu_l,\Sigma_l)}}$$

where $\mathcal{N}(x;\mu,\Sigma) = \frac{1}{(2\pi)^{d/2}|\Sigma|^{1/2}}\exp\!\left(-\tfrac{1}{2}(x-\mu)^T\Sigma^{-1}(x-\mu)\right)$.
\end{answer}

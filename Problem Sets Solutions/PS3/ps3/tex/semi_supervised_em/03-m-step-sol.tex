\begin{answer}
\textbf{Parameters to re-estimate:} $\phi_j$, $\mu_j$, $\Sigma_j$ for $j=1,\ldots,k$.

The M-step maximizes:
$$\mathcal{J} = \sum_{i=1}^\nexp\sum_{\zsi} w_{ij}\log\frac{p(\xsi,\zsi;\theta)}{w_{ij}} + \alpha\sum_{i=1}^{\tilde{\nexp}}\log p(\tilde{x}^{(i)},\tilde{z}^{(i)};\theta).$$

The relevant (parameter-dependent) terms are:
\begin{align*}
\mathcal{J} &\propto \sum_{i=1}^\nexp\sum_{j=1}^k w_{ij}\left[\log\phi_j + \log\mathcal{N}(\xsi;\mu_j,\Sigma_j)\right] \\
&\quad + \alpha\sum_{i=1}^{\tilde{\nexp}}\left[\log\phi_{\tilde{z}^{(i)}} + \log\mathcal{N}(\tilde{x}^{(i)};\mu_{\tilde{z}^{(i)}},\Sigma_{\tilde{z}^{(i)}})\right].
\end{align*}

\textbf{Update for $\phi_j$:} Using a Lagrange multiplier for $\sum_j\phi_j=1$ and differentiating:
$$\boxed{\phi_j^{(t+1)} = \frac{\displaystyle\sum_{i=1}^\nexp w_{ij} + \alpha\sum_{i=1}^{\tilde{\nexp}}\mathbf{1}\{\tilde{z}^{(i)}=j\}}{\nexp + \alpha\tilde{\nexp}}}$$

\textbf{Update for $\mu_j$:} Differentiating $\mathcal{J}$ w.r.t.\ $\mu_j$ and setting to zero:
$$\boxed{\mu_j^{(t+1)} = \frac{\displaystyle\sum_{i=1}^\nexp w_{ij}\,\xsi + \alpha\sum_{i=1}^{\tilde{\nexp}}\mathbf{1}\{\tilde{z}^{(i)}=j\}\,\tilde{x}^{(i)}}{\displaystyle\sum_{i=1}^\nexp w_{ij} + \alpha\sum_{i=1}^{\tilde{\nexp}}\mathbf{1}\{\tilde{z}^{(i)}=j\}}}$$

\textbf{Update for $\Sigma_j$:} Differentiating $\mathcal{J}$ w.r.t.\ $\Sigma_j^{-1}$ (or $\Sigma_j$) and setting to zero:
$$\boxed{\Sigma_j^{(t+1)} = \frac{\displaystyle\sum_{i=1}^\nexp w_{ij}(\xsi-\mu_j)(\xsi-\mu_j)^T + \alpha\sum_{i=1}^{\tilde{\nexp}}\mathbf{1}\{\tilde{z}^{(i)}=j\}(\tilde{x}^{(i)}-\mu_j)(\tilde{x}^{(i)}-\mu_j)^T}{\displaystyle\sum_{i=1}^\nexp w_{ij} + \alpha\sum_{i=1}^{\tilde{\nexp}}\mathbf{1}\{\tilde{z}^{(i)}=j\}}}$$

Note: $\mu_j^{(t+1)}$ in the $\Sigma_j$ update refers to the newly computed mean.
\end{answer}
